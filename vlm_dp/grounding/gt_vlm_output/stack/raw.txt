# stack -- deterministic stand-in for a GPT-4o ReKep response.
# Format follows rekep/prompts/prompt_template.txt, so _render() parses it exactly as
# rekep/constraint_generation.py parses a live response. Placeholders in braces are the values only
# resolvable at runtime: keypoint indices from the live scene and offsets measured from the
# observed point cloud.
#
# Task: "stack the small cube on the big cube"
#
# The task involves four stages:
# 1. Grasp cubeA.
# 2. Lift cubeA clear of the table.
# 3. Carry cubeA above cubeB.
# 4. Lower cubeA onto cubeB and release.
#
# Every target that the carried cube must reach is written as an offset from cubeB (keypoint
# {b}), which is the static fixture here. Anchoring on cubeA's own keypoint would be degenerate:
# the target would ride the object it is supposed to move.
#
# Transport is split from placement so that no rule is conditional: stage 3 holds the carry
# altitude, stage 4 gives it up and descends while staying over cubeB.

num_stages = 4


### stage 1 sub-goal constraints
def stage1_subgoal_constraint1(end_effector, keypoints):
    """Align the end-effector with cubeA (keypoint {a})."""
    return np.linalg.norm(end_effector - keypoints[{a}])

### stage 1 path constraints
# No path constraints needed for grasping.

### stage 2 sub-goal constraints
def stage2_subgoal_constraint1(end_effector, keypoints):
    """CubeA (keypoint {a}) is raised to its carry height above the pick-up pose."""
    offsetted_point = keypoints[{b}] + np.array({lift_a})
    return np.linalg.norm(keypoints[{a}] - offsetted_point)

### stage 2 path constraints
def stage2_path_constraint1(end_effector, keypoints):
    """CubeA (keypoint {a}) must not fall back below the height it was picked up from."""
    return np.maximum(0.0, keypoints[{b}][..., 2] + ({pick_z}) - keypoints[{a}][..., 2])

def stage2_path_constraint2(end_effector, keypoints):
    """The robot must still be holding cubeA (keypoint {a})."""
    return get_grasping_cost_by_keypoint_idx({a})

### stage 3 sub-goal constraints
def stage3_subgoal_constraint1(end_effector, keypoints):
    """CubeA (keypoint {a}) hovers above the stacking seat on cubeB (keypoint {b})."""
    offsetted_point = keypoints[{b}] + np.array({carry_a})
    return np.linalg.norm(keypoints[{a}] - offsetted_point)

### stage 3 path constraints
def stage3_path_constraint1(end_effector, keypoints):
    """CubeA (keypoint {a}) stays at transport altitude while crossing to cubeB."""
    return np.maximum(0.0, keypoints[{b}][..., 2] + ({carry_z}) - keypoints[{a}][..., 2])

def stage3_path_constraint2(end_effector, keypoints):
    """The robot must still be holding cubeA (keypoint {a})."""
    return get_grasping_cost_by_keypoint_idx({a})

### stage 4 sub-goal constraints
def stage4_subgoal_constraint1(end_effector, keypoints):
    """CubeA (keypoint {a}) rests on top of cubeB (keypoint {b})."""
    offsetted_point = keypoints[{b}] + np.array({seat_a})
    return np.linalg.norm(keypoints[{a}] - offsetted_point)

### stage 4 path constraints
def stage4_path_constraint1(end_effector, keypoints):
    """CubeA (keypoint {a}) stays over the seat on cubeB while it descends."""
    offsetted_point = keypoints[{b}] + np.array({seat_a})
    return np.linalg.norm(keypoints[{a}][..., :2] - offsetted_point[..., :2])

def stage4_path_constraint2(end_effector, keypoints):
    """The robot must still be holding cubeA (keypoint {a})."""
    return get_grasping_cost_by_keypoint_idx({a})


"""
Summarize keypoints to be grasped in all grasping stages.
For a grasping stage write the keypoint index; for a non-grasping stage write -1.
"""
grasp_keypoints = [{a}, -1, -1, -1]

"""
Summarize at **the end of which stage** the robot should release each keypoint.
"""
release_keypoints = [-1, -1, -1, {a}]
