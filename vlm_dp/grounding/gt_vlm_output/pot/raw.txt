# pot -- deterministic stand-in for a GPT-4o ReKep response.
# Format follows rekep/prompts/prompt_template.txt, so _render() parses it exactly as
# rekep/constraint_generation.py parses a live response. Placeholders in braces are the values only
# resolvable at runtime: keypoint indices from the live scene and offsets measured from the
# observed point cloud.
#
# Task: "remove the lid of the pot and put egg in it"
#
# The task involves eight stages:
# 1. Grasp the pot lid.
# 2. Lift the lid clear of the pot.
# 3. Carry the lid above its aside spot on the table.
# 4. Lower the lid onto the table.
# 5. Grasp the egg.
# 6. Lift the egg clear of the table.
# 7. Carry the egg above the pot rim.
# 8. Lower the egg onto the pot rim.
#
# Stages 2 and 6 are explicit lifts. They carry no grasp or release keypoint; their sub-goal
# constraint certifies the hold before transport begins. Without them the plan advances to
# transport on grasp proximity alone and discovers mid-carry that the hand is empty
# (measured on weight: 21 of 23 recoveries were "empty hand").
#
# BROUGHT TO PARITY WITH THE WEIGHT PLAN (the validated 16/20 reference). Three judgements,
# stated here because they are choices and not mechanical ports:
#
#  * TRANSPORT SPLIT FROM PLACEMENT (was 6 stages, now 8), for BOTH destinations -- the lid's
#    aside spot and the egg's pot rim. The old stages 3 and 6 each crossed the workspace and
#    descended in one stage, so a carry-height path rule on them would have forbidden their own
#    descent. Split, stages 3 and 7 carry an unconditional carry-height rule and stages 4 and 8
#    are pure descents. Exactly the structure weight uses for its two placements.
#
#  * ONE-SIDED VERTICAL LIFT SUB-GOALS (stages 2 and 6). Was a 3-D distance to a lift point
#    anchored on the pot keypoint. With subgoal_eps 0.12 a 3-D ball lets a lift complete at a
#    few centimetres of rise whenever the hand is xy-centred on the lift column; the vertical
#    shortfall against {lift_z_lid} / {lift_z_egg} makes "below eps" mean height and nothing
#    else, and stops penalising a lift that overshoots.
#
#  * GATED DESCENT ON BOTH LOWER STAGES (4 and 8). Their second path constraint forbids dropping
#    below the hover height while still horizontally far from the place point, relaxing linearly
#    to zero inside 8cm. The egg's 8cm gate radius is deliberately the SAME as the lid's and as
#    weight's, even though the pot mouth is narrower than 8cm across: the radius is the distance
#    at which the plan stops shaping the descent, not a claim about the opening's width, and one
#    number across every descent in every plan is worth more than a per-object tuning nobody has
#    measured.
#
# PLACE MODE UNCHANGED: stage 8's sub-goal deliberately avoids the words "into"/"inside"/
# "within", so rekep._places_into keeps compiling this placement as `surface`, exactly as the
# six-stage plan did. Restructuring is not the place to also flip an unmeasured cost mode.

num_stages = 8


### stage 1 sub-goal constraints
def stage1_subgoal_constraint1(end_effector, keypoints):
    """Align the end-effector with the pot lid (keypoint {lid})."""
    return np.linalg.norm(end_effector - keypoints[{lid}])

### stage 1 path constraints
# No path constraints needed for grasping.

### stage 2 sub-goal constraints
def stage2_subgoal_constraint1(end_effector, keypoints):
    """The lid (keypoint {lid}) has reached carry height (vertical shortfall only)."""
    return np.maximum(0.0, ({lift_z_lid}) - keypoints[{lid}][..., 2])

### stage 2 path constraints
def stage2_path_constraint1(end_effector, keypoints):
    """The robot must still be grasping the lid (keypoint {lid})."""
    return get_grasping_cost_by_keypoint_idx({lid})

### stage 3 sub-goal constraints
def stage3_subgoal_constraint1(end_effector, keypoints):
    """The lid (keypoint {lid}) hovers directly above its aside spot beside the pot (keypoint {pot})."""
    offsetted_point = keypoints[{pot}] + np.array({hover_lid})
    return np.linalg.norm(keypoints[{lid}] - offsetted_point)

### stage 3 path constraints
def stage3_path_constraint1(end_effector, keypoints):
    """The robot must still be grasping the lid (keypoint {lid})."""
    return get_grasping_cost_by_keypoint_idx({lid})

def stage3_path_constraint2(end_effector, keypoints):
    """Keep the lid (keypoint {lid}) clear of the pot and the table while carrying it across."""
    return np.maximum(0.0, ({carry_z_lid}) - keypoints[{lid}][..., 2])

### stage 4 sub-goal constraints
def stage4_subgoal_constraint1(end_effector, keypoints):
    """The lid (keypoint {lid}) rests on the table, clear of the pot (keypoint {pot})."""
    offsetted_point = keypoints[{pot}] + np.array({lid_off})
    return np.linalg.norm(keypoints[{lid}] - offsetted_point)

### stage 4 path constraints
def stage4_path_constraint1(end_effector, keypoints):
    """The robot must still be grasping the lid (keypoint {lid})."""
    return get_grasping_cost_by_keypoint_idx({lid})

def stage4_path_constraint2(end_effector, keypoints):
    """Stay above the hover height until horizontally aligned with the aside spot."""
    place_xy = keypoints[{pot}][..., :2] + np.array({lid_off})[:2]
    xy_off = np.linalg.norm(keypoints[{lid}][..., :2] - place_xy, axis=-1)
    low = np.maximum(0.0, ({hover_z_lid}) - keypoints[{lid}][..., 2])
    return low * np.minimum(1.0, xy_off / 0.08)

### stage 5 sub-goal constraints
def stage5_subgoal_constraint1(end_effector, keypoints):
    """Align the end-effector with the egg (keypoint {egg})."""
    return np.linalg.norm(end_effector - keypoints[{egg}])

### stage 5 path constraints
# No path constraints needed for grasping.

### stage 6 sub-goal constraints
def stage6_subgoal_constraint1(end_effector, keypoints):
    """The egg (keypoint {egg}) has reached carry height (vertical shortfall only)."""
    return np.maximum(0.0, ({lift_z_egg}) - keypoints[{egg}][..., 2])

### stage 6 path constraints
def stage6_path_constraint1(end_effector, keypoints):

    return get_grasping_cost_by_keypoint_idx({egg})

### stage 7 sub-goal constraints
def stage7_subgoal_constraint1(end_effector, keypoints):
    """The egg (keypoint {egg}) hovers directly above the pot rim (keypoint {pot})."""
    offsetted_point = keypoints[{pot}] + np.array({hover_egg})
    return np.linalg.norm(keypoints[{egg}] - offsetted_point)

### stage 7 path constraints
def stage7_path_constraint1(end_effector, keypoints):

def stage7_path_constraint2(end_effector, keypoints):
    """Keep the egg (keypoint {egg}) clear of the table while carrying it across."""

### stage 8 sub-goal constraints
def stage8_subgoal_constraint1(end_effector, keypoints):
    """The egg (keypoint {egg}) rests just above the pot rim (keypoint {pot})."""
    offsetted_point = keypoints[{pot}] + np.array({egg_off})
    return np.linalg.norm(keypoints[{egg}] - offsetted_point)

### stage 8 path constraints
def stage8_path_constraint1(end_effector, keypoints):

def stage8_path_constraint2(end_effector, keypoints):

"""
Summarize keypoints to be grasped in all grasping stages.
For a grasping stage write the keypoint index; for a non-grasping stage write -1.
"""
grasp_keypoints = [{lid}, -1, -1, -1, {egg}, -1, -1, -1]

"""
Summarize at **the end of which stage** the robot should release each keypoint.
"""
release_keypoints = [-1, -1, -1, {lid}, -1, -1, -1, {egg}]



# ---------------------------------------------------------------------------
# Completion predicates (vlm_dp extension to the ReKep response format).
#
# Written in the PRIMITIVE VOCABULARY the runtime provides (vlm_dp/grounding/predicates.py ::
# PredicateRuntime), exactly as the weight plan is. This plan states WHICH event each stage
# names; the primitive states HOW that event is sensed. Nothing here re-derives a window, a
# sensor threshold or a tolerance.
#
#   EVENTS, each of which also fills `components` with its named boolean and the margin_*
#   floats behind it:
#     grasped(kp, name, "acquire")   the pick-up event: fingers stalled on something for a
#                                    whole window, POSITIVE co-motion with a hand that
#                                    actually travelled, and the object within reach.
#     grasped(kp, name, "maintain")  still holding: fingers stalled and no slip relative to a
#                                    hand that moved. A still hand cannot slip, so it passes.
#     released(kp, name)             the hand let go: it departed while the object stayed, or
#                                    the fingers opened.
#     stationary(kp, name)           the object stopped moving -- "the lid is resting on the
#                                    table" is a statement about speed, which no distance to a
#                                    place point can see.
#     sustained(cost_fn, name)       this plan's OWN geometry, in the ReKep cost convention
#                                    (<= 0 satisfied, called per frame as cost_fn(ee, kp)),
#                                    held over the last few belief frames rather than at one
#                                    instant, so a single noisy sample cannot advance a stage.
#
#   TOLERANCES, resolved from the same grounding geometry the cost terms use:
#     target_region_radius(kp)  the placement region around an anchor keypoint. Both
#                               destinations anchor on the POT keypoint -- the lid's aside spot
#                               and the pot rim are both stated as offsets from it -- so both
#                               regions are asked for on {pot}.
#     clearance_margin(kp)      the vertical band separating "still clear above" from
#                               "descended onto". Asked for on the CARRIED body ({lid}, {egg}),
#                               which is what the primitive is defined for; asking for it on
#                               the pot would be refused, and correctly so -- a pot rim's
#                               clearance is not the lid's geometry.
#
#   DIAGNOSTIC KEYS. Every boolean these stages used to publish keeps its name and its meaning:
#   closed_on_object, rides_with_hand, at_hand, still_held, at_carry_height, inside_place_region,
#   still_clear_of_table (stage 3), still_clear_of_rim (stage 7), descended, hand_let_go,
#   object_quiescent, and the margins margin_aperture / margin_slip_ratio / margin_hand_disp_m /
#   margin_reach_m / margin_open_frac / margin_speed_mps. Stages 1 and 5 additionally publish an
#   overall lid_in_hand / egg_in_hand, which is the conjunction they used to return unnamed.
#   Three margins are RENAMED because sustained() reports one uniform quantity -- the worst COST
#   in the window, negative when satisfied -- where the old blocks reported differently-signed
#   ad-hoc numbers:
#     margin_height_shortfall_m -> margin_at_carry_height      (same value)
#     margin_xy_off_m           -> margin_inside_place_region  (now offset MINUS the radius)
#     margin_clearance_m        -> margin_still_clear_of_table / margin_still_clear_of_rim
#                                                              (now negated: cost, not clearance)
#     margin_descent_m          -> margin_descended            (now negated: cost, not clearance)
#
# The pot plan needs NO new primitive: its eight stages are two grasp / lift / transport / place
# quartets, so all eight predicates are the weight forms with the lid and the egg substituted for
# the pear and the apple. The only pot-specific wording is the two clearance diagnostics, which
# name what the carried body is being kept clear OF (the table for the lid, the rim for the egg).
#
# Interface: stage<N>_completion(history, end_effector, keypoints, robot_state, components)
# -> bool, filling `components` with named booleans plus margin_* floats. See
# vlm_dp/grounding/predicates.py for the full contract.
# ---------------------------------------------------------------------------


### stage 1 completion predicate
def stage1_completion(history, end_effector, keypoints, robot_state, components):
    """Lid grasped: the fingers are stalled on something, and the lid rides with the hand."""
    lid_in_hand = grasped({lid}, "lid_in_hand", "acquire")
    return bool(lid_in_hand)

### stage 2 completion predicate
def stage2_completion(history, end_effector, keypoints, robot_state, components):
    """Lid lifted: still owned by the hand, and it has held carry height for the last third of a second."""
    still_held = grasped({lid}, "still_held", "maintain")
    at_carry_height = sustained(
        lambda ee, kp: ({lift_z_lid}) - kp[{lid}][..., 2], "at_carry_height")
    return bool(still_held and at_carry_height)

### stage 3 completion predicate
def stage3_completion(history, end_effector, keypoints, robot_state, components):
    """Lid over its aside spot: still owned by the hand, inside the place region in xy, still clear above the table."""
    still_held = grasped({lid}, "still_held", "maintain")
    inside_place_region = sustained(
        lambda ee, kp: np.linalg.norm(kp[{lid}][..., :2] - (kp[{pot}][..., :2] + np.array({lid_off})[:2]),
                                      axis=-1) - target_region_radius({pot}), "inside_place_region")
    still_clear_of_table = sustained(
        lambda ee, kp: (({hover_z_lid}) - clearance_margin({lid})) - kp[{lid}][..., 2],
        "still_clear_of_table")
    return bool(still_held and inside_place_region and still_clear_of_table)

### stage 4 completion predicate
def stage4_completion(history, end_effector, keypoints, robot_state, components):
    """Lid placed: down in the place region, no longer owned by the hand, and no longer moving."""
    hand_let_go = released({lid}, "hand_let_go")
    inside_place_region = sustained(
        lambda ee, kp: np.linalg.norm(kp[{lid}][..., :2] - (kp[{pot}][..., :2] + np.array({lid_off})[:2]),
                                      axis=-1) - target_region_radius({pot}), "inside_place_region")
    object_quiescent = stationary({lid}, "object_quiescent")
    descended = sustained(
        lambda ee, kp: kp[{lid}][..., 2] - (({hover_z_lid}) - clearance_margin({lid})), "descended")
    return bool(hand_let_go and inside_place_region and object_quiescent and descended)
