# sort_can_tray_relational -- deterministic stand-in for a GPT-4o ReKep response.
# Format follows rekep/prompts/prompt_template.txt, so _render() parses it exactly as
# rekep/constraint_generation.py parses a live response. Placeholders in braces are the values only
# resolvable at runtime: keypoint indices from the live scene, the side the instruction asked for,
# and offsets measured from the observed geometry.
#
# Task: "place the can in the tray to the {side} of the marker by at least {margin} m"
#
# The semantic-programmability sibling of sort_can_tray. The scene is IDENTICAL and so are the
# grasp, lift and carry stages; only the terminal stage changes. Where sort_can_tray demanded a
# POINT, this program demands a RELATION: the can must end up on one named side of the marker,
# by at least {margin} m along the tray's x axis. Which side is not a property of the scene -- it
# is what the instruction said, and it enters this program as the sign {s}: +1 asks for the -x
# side ("left"), -1 asks for the +x side ("right"). The feasible set is a half-plane strip cut off
# by the tray walls, so it has area and no single position is privileged by the relation itself.
#
# The marker is the static fixture: keypoint {m} is its centre, and the relation is written
# against it. The can's own keypoint is {c}. Anchoring on {m} rather than on {c} matters -- a
# relation expressed relative to the payload would ride the payload and be satisfied everywhere.
#
# The task involves four stages:
# 1. Grasp the can.
# 2. Lift the can clear of the bin it starts in.
# 3. Carry the can over the tray, holding transport altitude so it clears the tray rim. The
#    transit waypoint is stated over the marker because the marker is what the relation is
#    measured from; the terminal stage, not this one, is what fixes where the can is set down.
# 4. Lower the can into the tray on the {side} side of the marker, and let go.
#
# Stage 4 states the objective in two parts, which the loader sums:
#   1. the SEMANTIC term -- a hinge that is exactly zero everywhere on the demanded side beyond
#      the {margin} m clearance and grows linearly as the relation is broken,
#   2. the TIE-BREAK term -- {w} times the horizontal distance from where the can was picked up,
#      which orders the (infinitely many) satisfying positions by how far the can has to travel.
# The tie-break weight is small enough that it can never buy a violation: the semantic term grows
# at one metre per metre of displacement across the boundary and the tie-break shrinks at most
# {w} metres per metre, so no point on the wrong side can undercut a point on the right one.

num_stages = 4


### stage 1 sub-goal constraints
def stage1_subgoal_constraint1(end_effector, keypoints):
    """Align the end-effector with the can (keypoint {c})."""
    return np.linalg.norm(end_effector - keypoints[{c}])

### stage 1 path constraints
# No path constraints needed for grasping.

### stage 2 sub-goal constraints
def stage2_subgoal_constraint1(end_effector, keypoints):
    """The can (keypoint {c}) is raised to its carry height above the pick-up pose."""
    offsetted_point = keypoints[{m}] + np.array({lift_c})
    return np.linalg.norm(keypoints[{c}] - offsetted_point)

### stage 2 path constraints
def stage2_path_constraint1(end_effector, keypoints):
    """The can (keypoint {c}) must not fall back below the height it was picked up from."""
    return np.maximum(0.0, keypoints[{m}][..., 2] + ({pick_z}) - keypoints[{c}][..., 2])

def stage2_path_constraint2(end_effector, keypoints):
    """The robot must still be holding the can (keypoint {c})."""
    return get_grasping_cost_by_keypoint_idx({c})

### stage 3 sub-goal constraints
def stage3_subgoal_constraint1(end_effector, keypoints):
    """The can (keypoint {c}) hovers over the marker (keypoint {m}), the relation's reference."""
    offsetted_point = keypoints[{m}] + np.array({carry_c})
    return np.linalg.norm(keypoints[{c}] - offsetted_point)

### stage 3 path constraints
def stage3_path_constraint1(end_effector, keypoints):
    """The can (keypoint {c}) stays at transport altitude, clear of the tray rim."""
    return np.maximum(0.0, keypoints[{m}][..., 2] + ({carry_z}) - keypoints[{c}][..., 2])

def stage3_path_constraint2(end_effector, keypoints):
    """The robot must still be holding the can (keypoint {c})."""
    return get_grasping_cost_by_keypoint_idx({c})

### stage 4 sub-goal constraints
def stage4_subgoal_constraint1(end_effector, keypoints):
    """The can (keypoint {c}) rests {side} of the marker (keypoint {m}) by at least {margin} m."""
    signed_gap = ({s}) * (keypoints[{c}][..., 0] - keypoints[{m}][..., 0])
    return np.maximum(0.0, signed_gap + ({margin}))

def stage4_subgoal_constraint2(end_effector, keypoints):
    """Tie-break: of the satisfying positions, prefer the one nearest where the can was picked up."""
    start_point = keypoints[{m}] + np.array({start_c})
    return ({w}) * np.linalg.norm(keypoints[{c}][..., :2] - start_point[..., :2])

### stage 4 path constraints
def stage4_path_constraint1(end_effector, keypoints):
    """The robot must still be holding the can (keypoint {c}) while it descends."""
    return get_grasping_cost_by_keypoint_idx({c})


"""
Summarize keypoints to be grasped in all grasping stages.

"""
