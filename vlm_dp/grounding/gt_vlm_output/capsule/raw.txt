# capsule -- deterministic stand-in for a GPT-4o ReKep response.
# Format follows rekep/prompts/prompt_template.txt, so _render() parses it exactly as
# rekep/constraint_generation.py parses a live response. Placeholders in braces are the values only
# resolvable at runtime: keypoint indices from the live scene and offsets measured from the
# observed point cloud.
#
# Task: "open the coffee maker lid and put the pod inside"
#
# The task involves six stages:
# 1. Grasp the coffee-maker lid.
# 2. Open the lid on its hinge.
# 3. Grasp the pod.
# 4. Lift the pod clear of the tray.
# 5. Carry the pod above the bay.
# 6. Lower the pod onto the bay opening.
#
# Stage 4 is an explicit lift, carrying no grasp or release keypoint: its sub-goal constraint
# certifies the pod is actually held before transport to the bay begins.
#
# BROUGHT TO PARITY WITH THE WEIGHT PLAN (the validated 16/20 reference). Four judgements,
# stated here because they are choices and not mechanical ports:
#
#  * TRANSPORT SPLIT FROM PLACEMENT (was 5 stages, now 6). The old stage 5 both crossed the
#    workspace and descended into the bay, so a carry-height rule on it would have forbidden the
#    descent it was supposed to perform. Splitting gives stage 5 an unconditional carry-height
#    path rule and stage 6 a pure descent -- the same reason weight splits 2-3 from 4.
#
#  * ONE-SIDED VERTICAL FOR BOTH RISING STAGES (2 and 4). Stage 4 is the ordinary lift and reads
#    exactly like weight stages 2/6. Stage 2 -- the lid opening -- is treated as a lift too,
#    which is the one non-obvious call here: the lip travels on a HINGE ARC, so it rises while
#    swinging back in xy, and a 3-D distance to the mid-air open point charges the plan for the
#    arc's horizontal component, which the mechanism dictates and the robot cannot choose. What
#    the stage actually names is "the lip has risen by the open lift", so the sub-goal is the
#    vertical shortfall of the lip against the declared open-goal keypoint and nothing else.
#    The open goal stays a KEYPOINT rather than becoming a baked scalar: it is unowned (it never
#    moves) and keeping it named is what keeps the declared point meaningful to the plan.
#    Stage 2's completion predicate is the matching LID-OPENED event. Note that stage 2 is a
#    RELEASE stage (release_keypoints[1] = the lip), so the runtime classifies it as a place
#    stage and, under advance.predicate_place_transitions, advances it on that predicate: the
#    predicate therefore states the event in the place vocabulary -- risen, hand let go, lid no
#    longer swinging -- and uses a 5cm band rather than a 5mm one, the same band weight's place
#    predicates use for their descent test, because a hinge arc does not stop on a millimetre.
#
#  * GATED DESCENT (stage 6). Its second path constraint forbids dropping below the hover height
#    while still horizontally far from the bay, relaxing linearly to zero inside 8cm. The pod
#    then arrives over the bay opening and goes straight down, instead of diving diagonally at
#    the machine and clipping its rim on the way in.
#
#  * PLACE MODE UNCHANGED. Stage 6's sub-goal deliberately avoids the words "into"/"inside"/
#    "within": rekep._places_into reads that text to choose container vs surface placement, and
#    this plan has always compiled as `surface`. Restructuring the plan is not the place to also
#    flip a cost mode that no run has measured.

num_stages = 6


### stage 1 sub-goal constraints
def stage1_subgoal_constraint1(end_effector, keypoints):
    """Align the end-effector with the lid lip (keypoint {lip})."""
    return np.linalg.norm(end_effector - keypoints[{lip}])

### stage 1 path constraints
# No path constraints needed for grasping.

### stage 2 sub-goal constraints
def stage2_subgoal_constraint1(end_effector, keypoints):
    """The lid lip (keypoint {lip}) has swung up to the open goal (keypoint {open_goal}) height."""
    return np.maximum(0.0, keypoints[{open_goal}][..., 2] - keypoints[{lip}][..., 2])

### stage 2 path constraints
def stage2_path_constraint1(end_effector, keypoints):
    """The robot must still be grasping the lid lip (keypoint {lip})."""
    return get_grasping_cost_by_keypoint_idx({lip})

### stage 3 sub-goal constraints
def stage3_subgoal_constraint1(end_effector, keypoints):
    """Align the end-effector with the pod (keypoint {pod})."""
    return np.linalg.norm(end_effector - keypoints[{pod}])

### stage 3 path constraints
# No path constraints needed for grasping.

### stage 4 sub-goal constraints
def stage4_subgoal_constraint1(end_effector, keypoints):
    """The pod (keypoint {pod}) has reached carry height (vertical shortfall only)."""
    return np.maximum(0.0, ({lift_z_pod}) - keypoints[{pod}][..., 2])

### stage 4 path constraints
def stage4_path_constraint1(end_effector, keypoints):
    """The robot must still be holding the pod (keypoint {pod})."""
    return get_grasping_cost_by_keypoint_idx({pod})

### stage 5 sub-goal constraints
def stage5_subgoal_constraint1(end_effector, keypoints):
    """The pod (keypoint {pod}) hovers directly above the bay (keypoint {bay})."""
    offsetted_point = keypoints[{bay}] + np.array({hover_pod})
    return np.linalg.norm(keypoints[{pod}] - offsetted_point)

### stage 5 path constraints
def stage5_path_constraint1(end_effector, keypoints):
    """The robot must still be holding the pod (keypoint {pod})."""
    return get_grasping_cost_by_keypoint_idx({pod})

def stage5_path_constraint2(end_effector, keypoints):
    """Keep the pod (keypoint {pod}) clear of the tray while carrying it across."""
    return np.maximum(0.0, ({carry_z_pod}) - keypoints[{pod}][..., 2])

### stage 6 sub-goal constraints
def stage6_subgoal_constraint1(end_effector, keypoints):
    """The pod (keypoint {pod}) rests at the bay opening (keypoint {bay})."""
    offsetted_point = keypoints[{bay}] + np.array({off_pod})
    return np.linalg.norm(keypoints[{pod}] - offsetted_point)

### stage 6 path constraints
def stage6_path_constraint1(end_effector, keypoints):
    """The robot must still be grasping the pod (keypoint {pod})."""
    return get_grasping_cost_by_keypoint_idx({pod})

def stage6_path_constraint2(end_effector, keypoints):
    """Stay above the hover height until horizontally aligned with the bay opening."""
    place_xy = keypoints[{bay}][..., :2] + np.array({off_pod})[:2]
    xy_off = np.linalg.norm(keypoints[{pod}][..., :2] - place_xy, axis=-1)
    low = np.maximum(0.0, ({hover_z_pod}) - keypoints[{pod}][..., 2])
    return low * np.minimum(1.0, xy_off / 0.08)


"""
Summarize keypoints to be grasped in all grasping stages.
For a grasping stage write the keypoint index; for a non-grasping stage write -1.
"""
grasp_keypoints = [{lip}, -1, {pod}, -1, -1, -1]

"""
Summarize at **the end of which stage** the robot should release each keypoint.
"""
release_keypoints = [-1, {lip}, -1, -1, -1, {pod}]



# ---------------------------------------------------------------------------
# Completion predicates (vlm_dp extension to the ReKep response format).
#
# Same vocabulary and same interface as the weight plan documents: CONTACT (the finger joint
# angle stalled in the band (0.08, 0.60) for a whole window, which a close on empty air can
# never satisfy), CO-MOTION as a dimensionless RATIO (slip / hand_disp < 0.5, evaluated only
# when the hand actually moved), SUSTAIN (five frames, not one instant), QUIESCENCE (< 0.015
# m/s -- "it is resting in the bay" is a statement about speed, which a distance cannot see),
# and REGION + ONE-SIDED VERTICAL rather than a 3-D ball. None of them re-expresses
# `subgoal < eps`: no predicate here measures a distance to its stage's sub-goal point.
#
# The capsule-specific event is LID-OPENED (stage 2): the lip has RISEN by the open lift and
# held that height, while the fingers are still stalled on it. Height, not arc position: the
# hinge chooses where the lip ends up in xy, so only the rise is evidence the robot produced.
#
# Interface: stage<N>_completion(history, end_effector, keypoints, robot_state, components)
# -> bool, filling `components` with named booleans plus margin_* floats. See
# vlm_dp/grounding/predicates.py for the full contract.
# ---------------------------------------------------------------------------


### stage 1 completion predicate
def stage1_completion(history, end_effector, keypoints, robot_state, components):
    """Lid lip grasped: the fingers are stalled on something, and the lip rides with the hand."""
    ap = np.asarray(history["gripper_aperture"], dtype=np.float64)
    eef = np.asarray(history["eef"], dtype=np.float64)
    obj = np.asarray([f[{lip}] for f in history["kp"]], dtype=np.float64)
    w = int(min(len(ap), len(eef), len(obj), 8))
    closed = bool(w >= 8 and np.all(ap[-w:] > 0.08) and np.all(ap[-w:] < 0.60))
    hand = float(np.linalg.norm(eef[-1] - eef[-w])) if w > 1 else 0.0
    slip = float(np.linalg.norm((obj[-1] - eef[-1]) - (obj[-w] - eef[-w]))) if w > 1 else 9.9
    ratio = slip / hand if hand > 1e-9 else 9.9
    carried = bool(w >= 8 and hand > 0.02 and ratio < 0.5)
    reach = float(np.linalg.norm(obj[-1] - eef[-1]))
    components["closed_on_object"] = closed
    components["rides_with_hand"] = carried
    components["at_hand"] = bool(reach < 0.10)
    components["margin_aperture"] = float(ap[-1])
    components["margin_slip_ratio"] = ratio
    components["margin_hand_disp_m"] = hand
    components["margin_reach_m"] = reach
    return bool(closed and carried and reach < 0.10)

### stage 2 completion predicate
def stage2_completion(history, end_effector, keypoints, robot_state, components):
    """Lid opened: the lip has risen by the open lift and stayed there, the hand let go, nothing moving."""
    ap = np.asarray(history["gripper_aperture"], dtype=np.float64)
    eef = np.asarray(history["eef"], dtype=np.float64)
    obj = np.asarray([f[{lip}] for f in history["kp"]], dtype=np.float64)
    dt = max(float(history["dt"]), 1e-6)
    w = int(min(len(ap), len(eef), len(obj), 8))
    k = int(min(len(obj), 5))
    rise = float(np.min(obj[-k:][..., 2]) - (keypoints[{open_goal}][..., 2] - 0.05))
    risen = bool(k >= 5 and rise >= 0.0)
    hand = float(np.linalg.norm(eef[-1] - eef[-w])) if w > 1 else 0.0
    slip = float(np.linalg.norm((obj[-1] - eef[-1]) - (obj[-w] - eef[-w]))) if w > 1 else 0.0
    ratio = slip / hand if hand > 0.02 else 0.0
    departed = bool(w >= 8 and hand > 0.03 and ratio > 0.7)
    open_frac = float(np.mean(ap[-w:] <= 0.06)) if w >= 4 else 0.0
    released = bool(departed or open_frac >= 0.25)
    speed = float(np.linalg.norm(obj[-1] - obj[-w]) / ((w - 1) * dt)) if w > 1 else 9.9
    still = bool(w >= 8 and speed < 0.015)
    components["lip_risen"] = risen
    components["hand_let_go"] = released
    components["lid_quiescent"] = still
    components["margin_rise_m"] = rise
    components["margin_slip_ratio"] = ratio
    components["margin_open_frac"] = open_frac
    components["margin_speed_mps"] = speed
    return bool(risen and released and still)

### stage 4 completion predicate
def stage4_completion(history, end_effector, keypoints, robot_state, components):
    """Pod lifted: still owned by the hand, and it has held carry height for the last third of a second."""
    ap = np.asarray(history["gripper_aperture"], dtype=np.float64)
    eef = np.asarray(history["eef"], dtype=np.float64)
    obj = np.asarray([f[{pod}] for f in history["kp"]], dtype=np.float64)
    w = int(min(len(ap), len(eef), len(obj), 8))
    closed = bool(w >= 8 and np.all(ap[-w:] > 0.08) and np.all(ap[-w:] < 0.60))
    hand = float(np.linalg.norm(eef[-1] - eef[-w])) if w > 1 else 0.0
    slip = float(np.linalg.norm((obj[-1] - eef[-1]) - (obj[-w] - eef[-w]))) if w > 1 else 9.9
    ratio = slip / hand if hand > 0.02 else 0.0
    held = bool(closed and ratio < 0.5)
    k = int(min(len(obj), 5))
    shortfall = float(np.max(({lift_z_pod}) - obj[-k:][..., 2]))
    high = bool(k >= 5 and shortfall <= 0.005)
    components["still_held"] = held
    components["at_carry_height"] = high
    components["margin_aperture"] = float(ap[-1])
    components["margin_slip_ratio"] = ratio
    components["margin_height_shortfall_m"] = shortfall
    return bool(held and high)

### stage 5 completion predicate
def stage5_completion(history, end_effector, keypoints, robot_state, components):
    """Pod over the bay: still owned by the hand, inside the bay region in xy, still clear above it."""
    ap = np.asarray(history["gripper_aperture"], dtype=np.float64)
    eef = np.asarray(history["eef"], dtype=np.float64)
    obj = np.asarray([f[{pod}] for f in history["kp"]], dtype=np.float64)
    w = int(min(len(ap), len(eef), len(obj), 8))
    closed = bool(w >= 8 and np.all(ap[-w:] > 0.08) and np.all(ap[-w:] < 0.60))
    hand = float(np.linalg.norm(eef[-1] - eef[-w])) if w > 1 else 0.0
    slip = float(np.linalg.norm((obj[-1] - eef[-1]) - (obj[-w] - eef[-w]))) if w > 1 else 9.9
    ratio = slip / hand if hand > 0.02 else 0.0
    held = bool(closed and ratio < 0.5)
    place_xy = keypoints[{bay}][..., :2] + np.array({off_pod})[:2]
    k = int(min(len(obj), 5))
    xy_off = float(np.max(np.linalg.norm(obj[-k:][..., :2] - place_xy, axis=-1)))
    over = bool(k >= 5 and xy_off < 0.08)
    clear = float(np.min(obj[-k:][..., 2]) - (({hover_z_pod}) - 0.05))
    above = bool(clear >= 0.0)
    components["still_held"] = held
    components["inside_place_region"] = over
    components["still_clear_of_bay"] = above
    components["margin_slip_ratio"] = ratio
    components["margin_xy_off_m"] = xy_off
    components["margin_clearance_m"] = clear
    return bool(held and over and above)

### stage 6 completion predicate
def stage6_completion(history, end_effector, keypoints, robot_state, components):
    """Pod placed: down in the bay region, no longer owned by the hand, and no longer moving."""
    ap = np.asarray(history["gripper_aperture"], dtype=np.float64)
    eef = np.asarray(history["eef"], dtype=np.float64)
    obj = np.asarray([f[{pod}] for f in history["kp"]], dtype=np.float64)
    dt = max(float(history["dt"]), 1e-6)
    w = int(min(len(ap), len(eef), len(obj), 8))
    place_xy = keypoints[{bay}][..., :2] + np.array({off_pod})[:2]
    k = int(min(len(obj), 5))
    xy_off = float(np.max(np.linalg.norm(obj[-k:][..., :2] - place_xy, axis=-1)))
    seated = bool(k >= 5 and xy_off < 0.08)
    hand = float(np.linalg.norm(eef[-1] - eef[-w])) if w > 1 else 0.0
    slip = float(np.linalg.norm((obj[-1] - eef[-1]) - (obj[-w] - eef[-w]))) if w > 1 else 0.0
    ratio = slip / hand if hand > 0.02 else 0.0
    departed = bool(w >= 8 and hand > 0.03 and ratio > 0.7)
    open_frac = float(np.mean(ap[-w:] <= 0.06)) if w >= 4 else 0.0
    released = bool(departed or open_frac >= 0.25)
    speed = float(np.linalg.norm(obj[-1] - obj[-w]) / ((w - 1) * dt)) if w > 1 else 9.9
    still = bool(w >= 8 and speed < 0.015)
    descend = float((({hover_z_pod}) - 0.05) - np.max(obj[-k:][..., 2]))
    down = bool(descend >= 0.0)
    components["inside_place_region"] = seated
    components["hand_let_go"] = released
    components["object_quiescent"] = still
    components["descended"] = down
    components["margin_xy_off_m"] = xy_off
    components["margin_slip_ratio"] = ratio
    components["margin_open_frac"] = open_frac
    components["margin_speed_mps"] = speed
    components["margin_descent_m"] = descend
    return bool(seated and released and still and down)
