# can -- deterministic stand-in for a GPT-4o ReKep response.
# Format follows rekep/prompts/prompt_template.txt, so _render() parses it exactly as
# rekep/constraint_generation.py parses a live response. Placeholders in braces are the values only
# resolvable at runtime: keypoint indices from the live scene and offsets measured from the
# observed point cloud.
#
# Task: "put the can in the bin"
#
# The task involves four stages:
# 1. Grasp the can.
# 2. Lift the can clear of the table.
# 3. Carry the can above the bin.
# 4. Lower the can to the release point over the bin and let go.
#
# The bin is the static fixture: keypoint {d} is the release point above its mouth, and every
# target the carried can must reach is an offset from it. The can is released above the bin rather
# than lowered to its floor -- the walls are in the way of the gripper, so the last few centimetres
# are a drop, which is why stage 4 targets the release point itself.

num_stages = 4


### stage 1 sub-goal constraints
def stage1_subgoal_constraint1(end_effector, keypoints):
    """Align the end-effector with the can (keypoint {c})."""
    return np.linalg.norm(end_effector - keypoints[{c}])

### stage 1 path constraints
# No path constraints needed for grasping.

### stage 2 sub-goal constraints
def stage2_subgoal_constraint1(end_effector, keypoints):
    """The can (keypoint {c}) is raised to its carry height above the pick-up pose."""
    offsetted_point = keypoints[{d}] + np.array({lift_c})
    return np.linalg.norm(keypoints[{c}] - offsetted_point)

### stage 2 path constraints
def stage2_path_constraint1(end_effector, keypoints):
    """The can (keypoint {c}) must not fall back below the height it was picked up from."""
    return np.maximum(0.0, keypoints[{d}][..., 2] + ({pick_z}) - keypoints[{c}][..., 2])

def stage2_path_constraint2(end_effector, keypoints):
    """The robot must still be holding the can (keypoint {c})."""
    return get_grasping_cost_by_keypoint_idx({c})

### stage 3 sub-goal constraints
def stage3_subgoal_constraint1(end_effector, keypoints):
    """The can (keypoint {c}) hovers above the bin release point (keypoint {d})."""
    offsetted_point = keypoints[{d}] + np.array({carry_c})
    return np.linalg.norm(keypoints[{c}] - offsetted_point)

### stage 3 path constraints
def stage3_path_constraint1(end_effector, keypoints):
    """The can (keypoint {c}) stays at transport altitude, clear of the bin rim."""
    return np.maximum(0.0, keypoints[{d}][..., 2] + ({carry_z}) - keypoints[{c}][..., 2])

def stage3_path_constraint2(end_effector, keypoints):
    """The robot must still be holding the can (keypoint {c})."""
    return get_grasping_cost_by_keypoint_idx({c})

### stage 4 sub-goal constraints
def stage4_subgoal_constraint1(end_effector, keypoints):
    """The can (keypoint {c}) reaches the release point over the bin (keypoint {d})."""
    return np.linalg.norm(keypoints[{c}] - keypoints[{d}])

### stage 4 path constraints
def stage4_path_constraint1(end_effector, keypoints):
    """The can (keypoint {c}) stays over the bin mouth while it descends."""
    return np.linalg.norm(keypoints[{c}][..., :2] - keypoints[{d}][..., :2])

def stage4_path_constraint2(end_effector, keypoints):
    """The robot must still be holding the can (keypoint {c})."""
    return get_grasping_cost_by_keypoint_idx({c})
